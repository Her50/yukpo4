**Accès**:

## 🔧 Guide d'Intégration

### Intégrer INFOODS (FAO)

```typescript
// 1. Télécharger fichiers CSV depuis le site FAO
// 2. Parser les données
// 3. Enrichir produits locaux avec données nutritionnelles

import { EnrichedProduct } from '../data/enrichedProductDatabase';

interface INFOODSData {
    produit: string;
    proteines: number;
    lipides: number;
    glucides: number;
    calories: number;
    vitamines: Record<string, number>;
}

function enrichWithINFOODS(product: EnrichedProduct, infoodsData: INFOODSData) {
    product.characteristics.caracteristiques_fixes = {
        ...product.characteristics.caracteristiques_fixes,
        valeur_nutritionnelle: {
            proteines: infoodsData.proteines,
            lipides: infoodsData.lipides,
            glucides: infoodsData.glucides,
            calories: infoodsData.calories,
            vitamines: infoodsData.vitamines
        }
    };
}
```
