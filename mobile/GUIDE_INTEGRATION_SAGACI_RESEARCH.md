## 💻 Intégration Technique

### Service d'Intégration Sagaci

Créer un service pour intégrer les données Sagaci :

```typescript
// mobile/src/services/sagaciProductService.ts

/**
 * Service d'intégration Sagaci Research
 * 
 * IMPORTANT: Nécessite clé API Sagaci (obtenue via contact commercial)
 */

interface SagaciConfig {
    apiKey: string;
    apiUrl: string;           // URL API Sagaci
    defaultCountry?: string;   // Pays par défaut
}

interface SagaciSearchOptions {
    country?: string;          // Code ISO pays (CM, CI, SN, etc.)
    city?: string;             // Ville spécifique
    category?: string;          // Catégorie produit
    searchQuery?: string;      // Recherche texte
    latitude?: number;         // Latitude pour recherche géographique
    longitude?: number;        // Longitude pour recherche géographique
    radiusKm?: number;         // Rayon de recherche (km)
    limit?: number;            // Nombre de résultats
}

class SagaciProductService {
    private config: SagaciConfig;
    private cache = new Map<string, any>();
    
    constructor(config: SagaciConfig) {
        this.config = config;
    }
    
    /**
     * Rechercher des produits avec filtrage géographique
     */
    async searchProducts(options: SagaciSearchOptions): Promise<SagaciProduct[]> {
        const cacheKey = JSON.stringify(options);
        
        // Vérifier le cache
        if (this.cache.has(cacheKey)) {
            return this.cache.get(cacheKey);
        }
        
        try {
            // Construire les paramètres de requête
            const params = new URLSearchParams();
            
            if (options.country) {
                params.append('country', options.country);
            }
            
            if (options.city) {
                params.append('city', options.city);
            }
            
            if (options.category) {
                params.append('category', options.category);
            }
            
            if (options.searchQuery) {
                params.append('q', options.searchQuery);
            }
            
            // Recherche géographique (si GPS fourni)
            if (options.latitude && options.longitude) {
                params.append('lat', options.latitude.toString());
                params.append('lng', options.longitude.toString());
                
                if (options.radiusKm) {
                    params.append('radius', options.radiusKm.toString());
                }
            }
            
            if (options.limit) {
                params.append('limit', options.limit.toString());
            }
            
            // Appel API Sagaci
            const response = await fetch(`${this.config.apiUrl}/products?${params}`, {
                headers: {
                    'Authorization': `Bearer ${this.config.apiKey}`,
                    'Content-Type': 'application/json'
                }
            });
            
            if (!response.ok) {
                throw new Error(`Sagaci API error: ${response.status}`);
            }
            
            const data = await response.json();
            const products = data.products || [];
            
            // Mettre en cache
            this.cache.set(cacheKey, products);
            
            return products;
            
        } catch (error) {
            console.error('[SagaciService] Erreur recherche:', error);
            return [];
        }
    }
    
    /**
     * Obtenir produits pour un pays spécifique
     */
    async getProductsByCountry(countryCode: string, category?: string): Promise<SagaciProduct[]> {
        return this.searchProducts({
            country: countryCode,
            category,
            limit: 1000
        });
    }
    
    /**
     * Recherche géographique (produits proches d'un point GPS)
     */
    async searchNearby(
        latitude: number,
        longitude: number,
        radiusKm: number = 20,
        category?: string
    ): Promise<SagaciProduct[]> {
        return this.searchProducts({
            latitude,
            longitude,
            radiusKm,
            category
        });
    }
    
    /**
     * Obtenir produits par ville
     */
    async getProductsByCity(
        countryCode: string,
        city: string,
        category?: string
    ): Promise<SagaciProduct[]> {
        return this.searchProducts({
            country: countryCode,
            city,
            category
        });
    }
}

// Instance singleton (à initialiser avec config)
export let sagaciProductService: SagaciProductService | null = null;

/**
 * Initialiser le service Sagaci
 */
export function initializeSagaciService(config: SagaciConfig): void {
    sagaciProductService = new SagaciProductService(config);
    console.log('[SagaciService] ✅ Service initialisé');
}

/**
 * Vérifier si le service est disponible
 */
export function isSagaciAvailable(): boolean {
    return sagaciProductService !== null;
}
```

### Transformer Données Sagaci → Format Yukpomnang

```typescript
// mobile/src/services/sagaciProductService.ts (suite)

import { EnrichedProduct } from '../data/enrichedProductDatabase';

/**
 * Transformer un produit Sagaci en format Yukpomnang
 */
function transformSagaciToYukpomnang(sagaciProduct: SagaciProduct): EnrichedProduct {
    return {
        id: `sagaci_${sagaciProduct.product_id}`,
        nom: sagaciProduct.name,
        aliases: [
            sagaciProduct.name,
            sagaciProduct.brand ? `${sagaciProduct.brand} ${sagaciProduct.name}` : undefined
        ].filter(Boolean) as string[],
        
        characteristics: {
            categorie: sagaciProduct.category,
            marque: sagaciProduct.brand,
            type: sagaciProduct.subcategory,
            unite: sagaciProduct.packaging_unit || 'unité',
            
            caracteristiques_fixes: {
                // Dimensions
                taille_emballage: sagaciProduct.packaging_size,
                poids: sagaciProduct.weight,
                volume: sagaciProduct.volume,
                
                // Géographie
                pays: sagaciProduct.country_name,
                region: sagaciProduct.region,
                ville: sagaciProduct.city,
                
                // Prix
                prix_reference: sagaciProduct.price,
                devise: sagaciProduct.currency,
                
                // Évaluations
                note_consommateurs: sagaciProduct.consumer_rating,
                nombre_avis: sagaciProduct.review_count,
                
                // Source
                source: 'sagaci',
                code_barre: sagaciProduct.barcode
            },
            
            caracteristiques_variables: [
                {
                    field: 'prix',
                    label: 'Prix',
                    type: 'number',
                    required: true
                },
                {
                    field: 'quantite',
                    label: 'Quantité',
                    type: 'number',
                    required: true
                }
            ]
        },
        
        // Disponibilité géographique
        availableIn: sagaciProduct.country_code ? [sagaciProduct.country_code] : undefined
    };
}

/**
 * Rechercher produits Sagaci et transformer au format Yukpomnang
 */
export async function searchSagaciProducts(
    query: string,
    countryCode: string,
    options?: {
        city?: string;
        category?: string;
        latitude?: number;
        longitude?: number;
        radiusKm?: number;
    }
): Promise<EnrichedProduct[]> {
    if (!sagaciProductService) {
        console.warn('[SagaciService] Service non initialisé');
        return [];
    }
    
    try {
        const sagaciProducts = await sagaciProductService.searchProducts({
            country: countryCode,
            city: options?.city,
            category: options?.category,
            searchQuery: query,
            latitude: options?.latitude,
            longitude: options?.longitude,
            radiusKm: options?.radiusKm,
            limit: 20
        });
        
        // Transformer en format Yukpomnang
        return sagaciProducts.map(transformSagaciToYukpomnang);
        
    } catch (error) {
        console.error('[SagaciService] Erreur recherche:', error);
        return [];
    }
}
```
